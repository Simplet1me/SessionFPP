cmake_minimum_required(VERSION 3.21)
project(SessionFPP LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
add_library(session_fpp SHARED native/mod.cpp native/settings.cpp)
target_include_directories(session_fpp PRIVATE third_party/ue4ss)
target_compile_definitions(session_fpp PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(session_fpp PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(session_fpp PRIVATE minhook "${CMAKE_SOURCE_DIR}/third_party/ue4ss/UE4SS.lib" user32)
set_target_properties(session_fpp PROPERTIES OUTPUT_NAME main)
include(CTest)
if(BUILD_TESTING)
    add_executable(settings_check native/settings_check.cpp native/settings.cpp)
    target_compile_definitions(settings_check PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(settings_check PRIVATE /utf-8 /W4 /EHsc)
    target_link_libraries(settings_check PRIVATE user32)
    add_test(NAME native_settings COMMAND settings_check)
endif()
if(BUILD_TESTING)
    add_executable(head_pose_check native/head_pose_check.cpp)
    target_compile_options(head_pose_check PRIVATE /utf-8 /W4 /EHsc)
    add_test(NAME head_pose COMMAND head_pose_check)
endif()
